def octal_to_string(o):
    return ''.join(chr(int(x, 8)) for x in o.split())

def DecodeDat(key, text):
    res = ""
    for i in range(len(text)):
        c = ord(text[i]) - ord('a')
        k = ord(key[i % len(key)]) - ord('a')
        res += chr((c - k) % 26 + ord('a'))
    return res
